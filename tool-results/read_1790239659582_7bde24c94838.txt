    28→/** Parse one SSE frame (event/data pair) from the stream. */
    29→export function parseSseFrame(frame: string): { event: string; data: string } | null {
    30→  let event = "message";
    31→  const dataLines: string[] = [];
    32→  for (const line of frame.split("\n")) {
    33→    if (line.startsWith("event:")) event = line.slice(6).trim();
    34→    else if (line.startsWith("data:")) dataLines.push(line.slice(5).trim());
    35→  }
    36→  if (dataLines.length === 0) return null;
    37→  return { event, data: dataLines.join("\n") };
    38→}

   287→/** Live view of a streaming engine at work (SSE stream consumer). */
   288→export function LiveStreamPanel({
   289→  stage,
   290→  text,
   291→  steps,
   292→  title,
   293→}: {
   294→  stage: StreamStage;
   295→  text: string;
   296→  /** Custom stage stepper (defaults to the answer-engine steps). */
   297→  steps?: { stage: Exclude<StreamStage, "idle" | "connecting">; labelKey: string; icon: LucideIcon }[];
   298→  /** Panel title (i18n key or literal); defaults to the answer-engine title. */
   299→  title?: string;
   300→}) {
   301→  const t = useT();
   302→  const scrollRef = useRef<HTMLDivElement>(null);
   303→
   304→  useEffect(() => {
   305→    const el = scrollRef.current;
   306→    if (el) el.scrollTop = el.scrollHeight;
   307→  }, [text]);
   308→
   309→  const activeSteps = steps ?? STREAM_STEPS;
   310→  const stepIndex = activeSteps.findIndex((s) => s.stage === stage);
   311→  const lines = text.split("\n");
   312→
   313→  return (
   314→    <Card className="border-emerald-300 shadow-[0_0_0_4px_rgba(16,185,129,0.06)] dark:border-emerald-800">
   315→      <CardContent className="p-5">
   316→        <div className="flex flex-wrap items-center justify-between gap-2">
   317→          <div className="flex items-center gap-2">
   318→            <span className="relative flex h-2.5 w-2.5" aria-hidden>
   319→              <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-500 opacity-60" />
   320→              <span className="relative inline-flex h-2.5 w-2.5 rounded-full bg-emerald-600" />
   321→            </span>
   322→            <h3 className="font-scholarly text-sm font-bold">{title ?? t("engineTitle")}</h3>
   323→            <Badge variant="outline" className="border-emerald-300 text-[9px] uppercase tracking-wider text-emerald-800 dark:border-emerald-800 dark:text-emerald-300">
   324→              <Radio className="me-1 h-2.5 w-2.5" aria-hidden /> {t("engineLive")}
   325→            </Badge>
   326→          </div>
   327→        </div>
   328→
   329→        {/* Stage stepper */}
   330→        <ol className="mt-3 flex flex-wrap items-center gap-x-4 gap-y-1.5" aria-label="Engine stages">
   331→          {activeSteps.map((s, i) => {
   332→            const isDone = stepIndex > i;
   333→            const isCurrent = stepIndex === i;
   334→            const Icon = s.icon;
   335→            return (
   336→              <li
   337→                key={s.stage}
   338→                aria-current={isCurrent ? "step" : undefined}
   339→                className={cn(
   340→                  "flex items-center gap-1.5 text-[11px] font-medium transition-colors",
   341→                  isDone ? "text-emerald-700 dark:text-emerald-400" : isCurrent ? "text-foreground" : "text-muted-foreground/60"
   342→                )}
   343→              >
   344→                {isDone ? (
   345→                  <Check className="h-3.5 w-3.5" aria-hidden />
   346→                ) : isCurrent ? (
   347→                  <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden />
   348→                ) : (
   349→                  <Icon className="h-3.5 w-3.5" aria-hidden />
   350→                )}
   351→                {t(s.labelKey as never)}
   352→              </li>
   353→            );
   354→          })}
   355→        </ol>
   356→
   357→        {/* Streaming text */}
   358→        <div
   359→          ref={scrollRef}
   360→          className="mt-3 max-h-80 overflow-y-auto rounded-lg border bg-emerald-50/40 p-3 dark:bg-emerald-950/20"
   361→          aria-live="polite"
   362→          aria-busy="true"
   363→        >
   364→          {text ? (
   365→            <div className="space-y-1 text-sm leading-relaxed">
   366→              {lines.map((line, i) => {
   367→                // Models may emit a heading inline ("…framing. ## Sources") — split it out.
   368→                const hi = line.indexOf("## ");
   369→                if (hi >= 0) {
   370→                  const before = line.slice(0, hi);
   371→                  const heading = line.slice(hi + 3);
   372→                  return (
   373→                    <p key={i} className="whitespace-pre-wrap">
   374→                      {before}
   375→                      <span className="block pt-2 text-[11px] font-bold uppercase tracking-wider text-emerald-800 dark:text-emerald-300">
   376→                        {heading}
   377→                      </span>
   378→                    </p>
   379→                  );
   380→                }
   381→                if (line.startsWith("## ")) {
   382→                  return (
   383→                    <p key={i} className="pt-2 text-[11px] font-bold uppercase tracking-wider text-emerald-800 dark:text-emerald-300">
   384→                      {line.slice(3)}
   385→                    </p>
   386→                  );
   387→                }
   388→                return <p key={i} className="whitespace-pre-wrap">{line}</p>;
   389→              })}
   390→              {stage === "generating" && <span className="stream-caret" aria-hidden />}
   391→            </div>
   392→          ) : (
   393→            <p className="text-sm italic text-muted-foreground">{stage === "saving" ? t("engineParsing") : t("engineHint")}</p>
   394→          )}
   395→        </div>
   396→      </CardContent>
   397→    </Card>
   398→  );

   401→/** Helpful / not-helpful voting — one vote per browser per answer. */
   402→export function FeedbackRow({ answerId, up, down }: { answerId: string; up: number; down: number }) {
   403→  const t = useT();
   404→  const { toast } = useToast();
   405→  const [mine, setMine] = useState<AnswerVote | null>(null);
   406→  const [counts, setCounts] = useState({ up, down });
   407→  const [pending, setPending] = useState(false);
   408→
   409→  useEffect(() => {
   410→    setMine(getVote(answerId));
   411→    setCounts({ up, down });
   412→  }, [answerId, up, down]);
   413→
   414→  const vote = async (v: Exclude<AnswerVote, null>) => {
   415→    if (pending || mine === v) return;
   416→    setPending(true);
   417→    const next = { up: counts.up + (v === "up" ? 1 : 0), down: counts.down + (v === "down" ? 1 : 0) };
   418→    setCounts(next); // optimistic
   419→    setVote(answerId, v);
   420→    setMine(v);
   421→    try {
   422→      const res = await apiPost<{ helpfulUp: number; helpfulDown: number }>("/api/answers/feedback", { answerId, vote: v });
   423→      setCounts({ up: res.helpfulUp, down: res.helpfulDown });
   424→      if (v === "up") toast({ title: t("thanksFeedback") });
   425→    } catch {
   426→      setCounts({ up, down }); // rollback on failure
   427→      setMine(getVote(answerId));
   428→    } finally {
   429→      setPending(false);
   430→    }
   431→  };
   432→
   433→  return (
   434→    <div className="flex items-center gap-1.5" role="group" aria-label={t("feedbackLabel")}>
   435→      <span className="me-1 hidden text-[11px] font-medium text-muted-foreground sm:inline">{t("helpful")}?</span>
   436→      <button
   437→        onClick={() => vote("up")}
   438→        disabled={pending}
   439→        aria-pressed={mine === "up"}
   440→        aria-label={t("helpful")}
   441→        title={t("helpful")}
   442→        className={cn(
   443→          "flex h-7 items-center gap-1 rounded-full border px-2.5 text-[11px] font-medium tabular-nums transition-all active:scale-90 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-emerald-600",
   444→          mine === "up"
   445→            ? "border-emerald-500 bg-emerald-50 text-emerald-800 dark:border-emerald-700 dark:bg-emerald-950 dark:text-emerald-300"
   446→            : "border-border text-muted-foreground hover:border-emerald-400 hover:text-emerald-700 dark:hover:text-emerald-300"
   447→        )}
   448→      >
   449→        <ThumbsUp className="h-3 w-3" aria-hidden /> {counts.up}
   450→      </button>
   451→      <button
   452→        onClick={() => vote("down")}
